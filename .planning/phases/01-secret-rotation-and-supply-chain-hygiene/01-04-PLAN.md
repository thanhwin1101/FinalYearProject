<objective>
Replace the removed-but-not-yet-replaced `OTA_PASSWORD` constant with per-device derivation: every ESP32 unit holds an identical deployment salt in NVS but derives a unique OTA password from its chip MAC. IT admin provisions the salt once per fleet via the deployment-salt.secret file and flashes each robot via `scripts/provision-esp32.ps1`. Closes SEC-04 fully.

Purpose: Reading one binary no longer leaks the OTA password for the whole fleet (the salt isn't compiled in), and stealing one robot's password doesn't give you flash access to the others (mac differs → derived password differs). OTA becomes a locked-per-unit capability.
Output: One provisioning script, one salt template, one new firmware module (provisioning console), a refactored ota_manager that performs HMAC derivation, a provisioning-mode entry path, and an OLED password-fingerprint display.

### mbedtls HMAC-SHA256 on ESP32 Arduino
Header: `#include "mbedtls/md.h"`
Usage:
```cpp
uint8_t mac[6];
esp_efuse_mac_get_default(mac);  // or ESP.getEfuseMac() → 48-bit integer
const mbedtls_md_info_t *info = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
mbedtls_md_context_t ctx;
mbedtls_md_init(&ctx);
mbedtls_md_setup(&ctx, info, 1 /*hmac*/);
mbedtls_md_hmac_starts(&ctx, salt, saltLen);
mbedtls_md_hmac_update(&ctx, mac, 6);
uint8_t out[32];
mbedtls_md_hmac_finish(&ctx, out);
mbedtls_md_free(&ctx);
// Base64url-encode `out`, truncate to 16 chars.
```

Base64url encoding (Arduino): `#include <base64.h>` from ESP32 Arduino core uses standard base64 (`+/=`). To get URL-safe, translate `+`→`-`, `/`→`_`, strip `=`. Alternative: use `mbedtls_base64_encode` from `mbedtls/base64.h`.

### Serial0 for provisioning console
`Serial` (USB-CDC via CH340 / CP210x) at 115200 baud — same port PlatformIO uses for upload/monitor. This is DIFFERENT from `Serial2` (UART to STM32) and `Serial1` (HuskyLens, disabled now). Safe to use without touching ADR-002.

### deployment-salt.secret file format
Single line, base64 of 32 random bytes (≈ 44 chars), no trailing newline. Generated once per fleet deployment. Same salt for every robot in the same deployment. IT admin can generate via:
```powershell
./scripts/generate-secure-password.ps1 -Length 44 | Set-Content -NoNewline deployment-salt.secret
```
Or explicitly 32 raw bytes:
```powershell
[byte[]]$b = New-Object byte[] 32
[System.Security.Cryptography.RandomNumberGenerator]::Create().GetBytes($b)
[Convert]::ToBase64String($b) | Set-Content -NoNewline deployment-salt.secret
```

CONTEXT.md D-05 specifies Hybrid: salt is NOT compiled, stored in NVS per unit, read from a gitignored `deployment-salt.secret` file on IT admin's machine OR from env. CONTEXT.md also says "Claude's discretion" for the detail: auto-generate if no file, or require file.

Decision matters because:
- **File-required:** IT admin must `generate-secure-password.ps1 -Length 44 > deployment-salt.secret` before first robot. If they don't, provisioning fails loud. Safer (no accidental fleet split — robots in same deployment share the salt). Harder first-run UX.

Recommended default: **File-required** — the UX friction pays for itself by preventing a whole class of "wait, why can't I flash Robot 01 from this laptop?" incidents.

```cpp
#pragma once
// Provisioning console — NVS read/write commands over Serial (USB).
//
// Entered only when g_provisioningMode is true (set in main.cpp setup() by
// 3-second long-press of BOOT button during first 3 s of boot).
//
// Command set (ASCII, \n-terminated, case-sensitive):
//   NVS_WRITE <key>=<value>      → writes string to NVS namespace "robotcfg"
//   NVS_READ <key>               → prints "OK <value>" or "ERR not_found"
//   NVS_ERASE                    → clears entire NVS namespace
//   NVS_FINGERPRINT              → prints SHA256 fingerprint of
//                                  concatenated (ssid|pass|srv|port|user|pass|ota_salt)
//                                  → "OK <64-hex>". Does NOT reveal individual values.
//   EXIT                         → reboot; resumes normal boot flow.
//   HELP                         → prints this list.
//
// Valid keys for NVS_WRITE (whitelist): wifi_ssid, wifi_pass, mqtt_srv,
// mqtt_port, mqtt_user, mqtt_pass, ota_salt. Anything else → ERR bad_key.
//
// Blocking — does not return until operator sends EXIT (or watchdog resets).

void provisioningConsoleRun();
```

**`provisioning_console.cpp`** (new):

Implement the command loop. Key design points:

1. `void provisioningConsoleRun()`:
   ```cpp
   #include "provisioning_console.h"
   #include "config.h"
   #include "oled_display.h"
   #include <Preferences.h>
   #include <Arduino.h>
   #include <mbedtls/md.h>

   static const char *VALID_KEYS[] = {
       "wifi_ssid", "wifi_pass", "mqtt_srv", "mqtt_port",
       "mqtt_user", "mqtt_pass", "ota_salt"
   };
   static const int VALID_KEYS_N = 7;

   static bool isValidKey(const String &k) {
       for (int i = 0; i < VALID_KEYS_N; i++) {
           if (k == VALID_KEYS[i]) return true;
       }
       return false;
   }

   static void printHelp() {
       Serial.println(F("=== ESP32 Provisioning Console ==="));
       Serial.println(F("Commands (terminate each with newline):"));
       Serial.println(F("  NVS_WRITE <key>=<value>"));
       Serial.println(F("  NVS_READ <key>"));
       Serial.println(F("  NVS_ERASE"));
       Serial.println(F("  NVS_FINGERPRINT"));
       Serial.println(F("  EXIT"));
       Serial.println(F("  HELP"));
       Serial.println(F("Valid keys: wifi_ssid wifi_pass mqtt_srv mqtt_port mqtt_user mqtt_pass ota_salt"));
   }

   static void sha256Hex(const uint8_t *data, size_t len, char outHex[65]) {
       const mbedtls_md_info_t *info = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
       uint8_t d[32];
       mbedtls_md(info, data, len, d);
       static const char HEX[] = "0123456789abcdef";
       for (int i = 0; i < 32; i++) {
           outHex[i*2]   = HEX[d[i] >> 4];
           outHex[i*2+1] = HEX[d[i] & 0x0F];
       }
       outHex[64] = 0;
   }

   void provisioningConsoleRun() {
       oledError("PROVISIONING");   // distinct OLED so operator knows they're in this mode
       printHelp();
       Serial.println(F("Enter command:"));

       while (true) {
           esp_task_wdt_reset();
           if (!Serial.available()) { delay(5); continue; }
           String line = Serial.readStringUntil('\n');
           line.trim();
           if (line.length() == 0) continue;

           if (line == "HELP") { printHelp(); continue; }
           if (line == "EXIT") { Serial.println(F("OK reboot")); delay(100); ESP.restart(); }
           if (line == "NVS_ERASE") {
               Preferences p; p.begin(NVS_NAMESPACE, false); p.clear(); p.end();
               Serial.println(F("OK erased"));
               continue;
           }
           if (line == "NVS_FINGERPRINT") {
               Preferences p; p.begin(NVS_NAMESPACE, true);
               String s = p.getString("wifi_ssid", "") + "|"
                        + p.getString("wifi_pass", "") + "|"
                        + p.getString("mqtt_srv",  "") + "|"
                        + String((uint16_t)p.getUShort("mqtt_port", 0)) + "|"
                        + p.getString("mqtt_user", "") + "|"
                        + p.getString("mqtt_pass", "") + "|"
                        + p.getString("ota_salt",  "");
               p.end();
               char hex[65];
               sha256Hex((const uint8_t*)s.c_str(), s.length(), hex);
               Serial.print(F("OK "));
               Serial.println(hex);
               continue;
           }
           if (line.startsWith("NVS_READ ")) {
               String key = line.substring(9); key.trim();
               if (!isValidKey(key)) { Serial.println(F("ERR bad_key")); continue; }
               Preferences p; p.begin(NVS_NAMESPACE, true);
               if (key == "mqtt_port") {
                   uint16_t v = p.getUShort("mqtt_port", 0);
                   p.end();
                   if (v == 0) Serial.println(F("ERR not_found"));
                   else { Serial.print(F("OK ")); Serial.println(v); }
               } else {
                   String v = p.getString(key.c_str(), "");
                   p.end();
                   if (v.length() == 0) Serial.println(F("ERR not_found"));
                   else { Serial.print(F("OK ")); Serial.println(v); }
               }
               continue;
           }
           if (line.startsWith("NVS_WRITE ")) {
               String body = line.substring(10);
               int eq = body.indexOf('=');
               if (eq < 0) { Serial.println(F("ERR missing_eq")); continue; }
               String key = body.substring(0, eq); key.trim();
               String val = body.substring(eq + 1);
               if (!isValidKey(key)) { Serial.println(F("ERR bad_key")); continue; }
               if (val.length() > 96) { Serial.println(F("ERR value_too_long")); continue; }
               Preferences p; p.begin(NVS_NAMESPACE, false);
               bool ok;
               if (key == "mqtt_port") {
                   uint16_t port = (uint16_t)val.toInt();
                   if (port == 0) { p.end(); Serial.println(F("ERR invalid_port")); continue; }
                   ok = p.putUShort("mqtt_port", port) > 0;
               } else {
                   ok = p.putString(key.c_str(), val) > 0;
               }
               p.end();
               Serial.println(ok ? F("OK") : F("ERR write_failed"));
               continue;
           }
           Serial.println(F("ERR unknown_cmd"));
       }
   }
   ```

```cpp
// Base64 decode helper
static int base64Decode(const String &in, uint8_t *out, size_t outCap) {
    size_t written = 0;
    int rc = mbedtls_base64_decode(out, outCap, &written,
                                   (const unsigned char *)in.c_str(), in.length());
    return (rc == 0) ? (int)written : -1;
}

// Base64url encode: standard base64 with +→-, /→_, strip padding.
static String base64UrlEncode(const uint8_t *in, size_t inLen) {
    size_t outCap = ((inLen + 2) / 3) * 4 + 1;
    uint8_t *buf = (uint8_t *)malloc(outCap);
    if (!buf) return String();
    size_t written = 0;
    int rc = mbedtls_base64_encode(buf, outCap, &written, in, inLen);
    String s;
    if (rc == 0) {
        s.reserve(written);
        for (size_t i = 0; i < written; i++) {
            char c = (char)buf[i];
            if (c == '+') c = '-';
            else if (c == '/') c = '_';
            else if (c == '=') continue;
            s += c;
        }
    }
    free(buf);
    return s;
}

bool otaDerivePassword(char *pwdOut, size_t bufLen) {
    if (bufLen < 17) return false;

    // Load salt from NVS
    Preferences prefs;
    prefs.begin(NVS_NAMESPACE, true);
    String saltB64 = prefs.getString("ota_salt", "");
    prefs.end();
    if (saltB64.length() == 0) return false;

    // Decode base64 salt → raw bytes
    uint8_t saltRaw[64];
    int saltLen = base64Decode(saltB64, saltRaw, sizeof(saltRaw));
    if (saltLen <= 0 || saltLen < 16) {
        Serial.printf("[OTA] salt decode failed or too short (%d)\n", saltLen);
        return false;
    }

    // Chip MAC (6 bytes)
    uint64_t macInt = ESP.getEfuseMac();
    uint8_t mac[6];
    for (int i = 0; i < 6; i++) mac[i] = (macInt >> (8 * i)) & 0xFF;

    // HMAC-SHA256(salt, mac)
    uint8_t hmacOut[32];
    const mbedtls_md_info_t *info = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
    mbedtls_md_context_t ctx;
    mbedtls_md_init(&ctx);
    if (mbedtls_md_setup(&ctx, info, 1) != 0) { mbedtls_md_free(&ctx); return false; }
    mbedtls_md_hmac_starts(&ctx, saltRaw, saltLen);
    mbedtls_md_hmac_update(&ctx, mac, 6);
    mbedtls_md_hmac_finish(&ctx, hmacOut);
    mbedtls_md_free(&ctx);

    // Base64url-encode → truncate to 16 chars
    String pwd = base64UrlEncode(hmacOut, 32);
    if (pwd.length() < 16) return false;
    strncpy(pwdOut, pwd.substring(0, 16).c_str(), bufLen - 1);
    pwdOut[bufLen - 1] = 0;
    return true;
}

bool otaPasswordFingerprint(const char *pwd, char *fingerprintOut, size_t bufLen) {
    if (bufLen < 7) return false;
    const mbedtls_md_info_t *info = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
    uint8_t d[32];
    mbedtls_md(info, (const uint8_t*)pwd, strlen(pwd), d);
    static const char HEX[] = "0123456789abcdef";
    for (int i = 0; i < 3; i++) {
        fingerprintOut[i*2]   = HEX[d[i] >> 4];
        fingerprintOut[i*2+1] = HEX[d[i] & 0x0F];
    }
    fingerprintOut[6] = 0;
    return true;
}

void otaInit() {
    char pwd[17];
    if (!otaDerivePassword(pwd, sizeof(pwd))) {
        Serial.println("[BOOT_WARN] ota_disabled");
        Serial.println("[OTA] ota_salt missing or invalid — OTA disabled");
        oledError("OTA DISABLED");
        return;
    }

    char fp[7];
    otaPasswordFingerprint(pwd, fp, sizeof(fp));
    Serial.printf("[OTA] derived password fingerprint: %s\n", fp);
    oledOtaFingerprint(fp);   // 2-second display
    delay(2000);              // hold it so operator can read

    ArduinoOTA.setHostname(OTA_HOSTNAME);
    ArduinoOTA.setPort(3232);
    ArduinoOTA.setPassword(pwd);   // per-device, derived
```
